#### SC virus mapping
module load Bowtie2 SAMtools
module load Subread

REF=GCF_000847405.1_ViralProj14643_genomic.fna
SAMDIR=368901
SAMPLE=368901

bowtie2-build "$REF" LBC_index
bowtie2 -x LBC_index -1 ${SAMDIR}_1_cutadapt.fastq.gz -2 ${SAMDIR}_1_cutadapt.fastq.gz --very-sensitive --end-to-end --dovetail --no-discordant -p 8 2> ${SAMPLE}.LBC.log | samtools sort -@ 8 -o ${SAMPLE}.LBC.sorted.bam

samtools index ${SAMPLE}.LBC.sorted.bam
samtools view -bh -q 10 -F 2308 ${SAMPLE}.LBC.sorted.bam > ${SAMPLE}.LBC.q10.bam
samtools index ${SAMPLE}.LBC.q10.bam
samtools depth -aa ${SAMPLE}.LBC.q10.bam > ${SAMPLE}.LBC.depth.tsv
